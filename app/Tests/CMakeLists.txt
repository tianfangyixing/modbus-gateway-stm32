cmake_minimum_required(VERSION 3.24)
project(configuration_host_tests C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(CONFIGURATION_TEST_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(CONFIGURATION_TEST_SANITIZERS)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
        message(FATAL_ERROR "Sanitizer tests require GCC/Clang on a supported non-Windows host")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(common)
add_subdirectory(configuration_codec)
add_subdirectory(configuration_service)
add_subdirectory(management)
add_subdirectory(mqtt)
